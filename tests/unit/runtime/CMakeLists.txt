# Runtime unit tests (WO-076, SL-ARCH-003 ARCH3-R014/R016). Every tests/unit/runtime/<stem>_test.cpp belongs to the area
# <stem> up to its first '_' (err_codes_test -> err). One doctest executable per area, rt_unit_<area>, and one ctest entry
# per file, runtime_<stem>, running only that file's cases (--source-file); labels unit, runtime and runtime_<area>
# (`just runtime-test <area>`). Areas stress and gate run serially under RESOURCE_LOCK cpu; sanitizer builds scale them by
# SL_RUNTIME_STRESS_SCALE=0.1. Later work orders only add files. Nobody edits this file again.
set(_sl_rt_tier unit)
if(NOT TARGET sl_runtime_all_tests)
  # The scripts build only this target (ARCH3-R016): every rt_* executable and the libraries they test.
  add_custom_target(sl_runtime_all_tests)
  add_dependencies(sl_runtime_all_tests sl_runtime sl_runtime_testing)
endif()
file(GLOB _sl_rt_tests CONFIGURE_DEPENDS "${CMAKE_CURRENT_SOURCE_DIR}/*_test.cpp")
set(_sl_rt_areas "")
foreach(src IN LISTS _sl_rt_tests)
  get_filename_component(stem "${src}" NAME_WE)
  string(REGEX REPLACE "_.*$" "" area "${stem}")
  list(APPEND _sl_rt_areas "${area}")
  list(APPEND _sl_rt_sources_${area} "${src}")
endforeach()
list(REMOVE_DUPLICATES _sl_rt_areas)
foreach(area IN LISTS _sl_rt_areas)
  set(exe rt_${_sl_rt_tier}_${area})
  sl_require(doctest)
  set(main "${CMAKE_CURRENT_BINARY_DIR}/${exe}_doctest_main.cpp")
  file(GENERATE OUTPUT "${main}" CONTENT "#define DOCTEST_CONFIG_IMPLEMENT_WITH_MAIN\n#include <doctest/doctest.h>\n")
  add_executable(${exe} ${_sl_rt_sources_${area}} "${main}")
  target_link_libraries(${exe} PRIVATE sl::runtime sl::runtime_testing sl::doctest)
  if(area STREQUAL "gate" AND TARGET sl_runtime_demo_scene)
    target_link_libraries(${exe} PRIVATE sl_runtime_demo_scene)
  endif()
  target_compile_definitions(${exe} PRIVATE SL_RUNTIME_SOURCE_DIR="${PROJECT_SOURCE_DIR}"
    SL_RUNTIME_EVIDENCE_ROOT="${PROJECT_SOURCE_DIR}/build/runtime")
  sl_target(${exe})
  add_dependencies(sl_runtime_all_tests ${exe})
  foreach(src IN LISTS _sl_rt_sources_${area})
    get_filename_component(stem "${src}" NAME_WE)
    set(name runtime_${stem})
    add_test(NAME ${name} COMMAND ${exe} "--source-file=*/${stem}.cpp")
    _sl_test_finish(${name} ${_sl_rt_tier} "")
    set_property(TEST ${name} APPEND PROPERTY LABELS runtime runtime_${area})
    if(SL_SANITIZE)
      set_property(TEST ${name} APPEND PROPERTY ENVIRONMENT "SL_RUNTIME_STRESS_SCALE=0.1")
    endif()
    if(area STREQUAL "stress" OR area STREQUAL "gate")
      set_tests_properties(${name} PROPERTIES RUN_SERIAL TRUE RESOURCE_LOCK cpu)
    endif()
  endforeach()
endforeach()
